🧬 The Cell at L3 (Harness) — Composition Example

cell ImageGenAgent
  // L1 Address
  addr  /agents/image-gen
  
  // L2 Scale
  scale 5
  
  // L3 Room + L4 🐘 Elephant
  room  cog-harness
  elephant field {
    warmth = +0.4
    kappa = 2.0
    dials { presence = 0.7, model_vs_code = +0.6 }
  }
  
  // L3 EXECUTOR substrate (lever-runner)
  executor  lever-runner {
    gates: [rust-fastloop, python-cache, llm]
    trust: 0.95
    surfaces: [cli, telegram, http]
  }
  
  // L3 MEMORY substrate (collective-unconscious)
  memory  collective-unconscious {
    vectors: [semantic, vibe, identity]
    temporal: [wall, phase, epoch, age, relationship]
    jepa: [growth, stuckness, direction, velocity, acceleration, novelty]
  }
  
  // L3 COMPILER substrate (forgemaster)
  compiler  forgemaster {
    constraints: { max_memory_mb: 8192, latency_ms: 5000 }
    proofs: [ptp-clock, heterogeneous-fleet]
  }
  
  // L3 DISCOVERY substrate (superinstance-agent)
  discovery  superinstance-agent {
    index: "fleet-crates"
    embedding: "bge-small-384"
    llm: "llama-3.1-8b"
  }
  
  // L3 COGNITION substrate (VaaS)
  cognition  vaas {
    shell: turbo  // GPU wheelhouse PC
    pillars: [thermodynamics, memory, timing, grafting]
    field: Ψ(t)
  }
  
  // L3 SPATIAL substrate (MUD family)
  spatial  mud-engine {
    rooms: 16
    npcs: enabled
    portal_to: "/spaces/prompt-room"
  }
  
  // L5 Protocol
  protocol  A2A
  
  // L6 Form
  form  ImageGenAgent {
    field prompt: str
    field negative: str
    field cfg: float = 7.5
    field steps: int = 30
    opener generate(prompt: str) -> Image
    opener teach(name: str, command: str) -> Unit  // via lever-runner
  }
  
  // L7 State
  state  { prompt: "a cat in a spacesuit", history: [..., {ts, image, rating}] }
end
